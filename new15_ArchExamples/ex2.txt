


import pyrtl

# ----------------------------
# Step 1: Define 1-bit Adder
# ----------------------------

def one_bit_add(a, b, carry_in):
    assert len(a) == len(b) == 1
    sum = a ^ b ^ carry_in
    carry_out = (a & b) | (a & carry_in) | (b & carry_in)
    return sum, carry_out

# ----------------------------
# Step 2: Define Ripple Carry Adder
# ----------------------------

def ripple_add(a, b, carry_in=0):
    a, b = pyrtl.match_bitwidth(a, b)

    if len(a) == 1:
        return one_bit_add(a, b, carry_in)

    lsb, ripple_carry = one_bit_add(a[0], b[0], carry_in)
    msbits, carry_out = ripple_add(a[1:], b[1:], ripple_carry)

    return pyrtl.concat(msbits, lsb), carry_out

# ----------------------------
# Step 3: Build Counter
# ----------------------------

counter = pyrtl.Register(bitwidth=3, name="counter")

# Add 1 each cycle using ripple carry adder
sum, carry_out = ripple_add(counter, pyrtl.Const("1'b1"))

counter.next <<= sum

# ----------------------------
# Step 4: Simulate
# ----------------------------

sim = pyrtl.Simulation()

for cycle in range(15):
    sim.step()
    assert sim.value[counter] == cycle % 8  # modulo-8 counter check

# Display waveform
sim.tracer.render_trace()

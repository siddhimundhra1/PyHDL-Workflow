# [Keyword]: Conditional Assignment State Machine (Vending Machine)

# [Design Category]: Sequential Logic

# [Design Function Description]:
# This circuit implements a finite state machine (FSM) for a vending machine.
# The machine dispenses an item after receiving 4 tokens. If a refund is requested,
# it returns the tokens. The design demonstrates conditional state transitions
# using `conditional_assignment`.

# [Input Signal Description]:
# token_in: 1-bit input indicating whether a token is inserted
# req_refund: 1-bit input requesting a refund

# [Output Signal Description]:
# dispense: 1-bit output indicating item is dispensed
# refund: 1-bit output indicating refund is issued

# [Design Detail]:
# States:
#   WAIT → TOK1 → TOK2 → TOK3 → DISP
#   Any state + refund request → RFND
#   DISP or RFND → WAIT
#
# dispense = (state == DISP)
# refund = (state == RFND)

import enum
import pyrtl

# ----------------------------
# Step 1: Define Inputs, Outputs, State
# ----------------------------

token_in = pyrtl.Input(1, "token_in")
req_refund = pyrtl.Input(1, "req_refund")

dispense = pyrtl.Output(1, "dispense")
refund = pyrtl.Output(1, "refund")

state = pyrtl.Register(3, "state")

# ----------------------------
# Step 2: Define States
# ----------------------------

class State(enum.IntEnum):
    WAIT = 0
    TOK1 = 1
    TOK2 = 2
    TOK3 = 3
    DISP = 4
    RFND = 5

# ----------------------------
# Step 3: Define FSM Transitions
# ----------------------------

with pyrtl.conditional_assignment:
    # Highest priority: refund request
    with req_refund:
        state.next |= State.RFND

    # Token insertion logic
    with token_in:
        with state == State.WAIT:
            state.next |= State.TOK1
        with state == State.TOK1:
            state.next |= State.TOK2
        with state == State.TOK2:
            state.next |= State.TOK3
        with state == State.TOK3:
            state.next |= State.DISP
        with pyrtl.otherwise:
            state.next |= State.RFND

    # Automatic reset transitions
    with (state == State.DISP) | (state == State.RFND):
        state.next |= State.WAIT

# ----------------------------
# Step 4: Output Logic
# ----------------------------

dispense <<= state == State.DISP
refund <<= state == State.RFND

# ----------------------------
# Step 5: Simulation
# ----------------------------

sim = pyrtl.Simulation()

sim_inputs = {
    "token_in": "0010100111010000",
    "req_refund": "1100010000000000"
}

sim.step_multiple(sim_inputs)

sim.tracer.render_trace(
    trace_list=["token_in", "req_refund", "state", "dispense", "refund"],
    repr_per_name={"state": pyrtl.enum_name(State)},
)

# ----------------------------
# Step 6: Verification
# ----------------------------

expected_sim_outputs = {
    "dispense": "0000000000001000",
    "refund": "0111001000000000"
}

sim = pyrtl.Simulation()
sim.step_multiple(sim_inputs, expected_sim_outputs)

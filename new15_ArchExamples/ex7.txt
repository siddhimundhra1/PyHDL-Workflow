# [Keyword]: PyRTL Timing and Optimization Example

# [Design Category]: Timing Analysis / Area Estimation / Circuit Optimization

# [Design Function Description]:
# This example demonstrates how to perform timing analysis, area estimation,
# synthesis, and logic optimization on a PyRTL hardware block. It shows how
# to inspect critical paths, estimate ASIC area, and reduce combinatorial
# logic complexity.

# [Input Signal Description]:
# input2: 4-bit input

# [Output Signal Description]:
# output: 5-bit output wire representing const_wire + input2

# [Design Detail]:
# - Timing analysis: measures maximum combinational path length
# - Area estimation: estimates logic and memory area in sq mm
# - Synthesis: breaks complex logic into 1-bit operations
# - Optimization: removes redundant logic

import pyrtl

# ----------------------------
# Part 0: Setup
# ----------------------------
pyrtl.reset_working_block()

const_wire = pyrtl.Const(6, bitwidth=4)
in_wire2 = pyrtl.Input(bitwidth=4, name="input2")
out_wire = pyrtl.Output(bitwidth=5, name="output")
out_wire <<= const_wire + in_wire2

# ----------------------------
# Part 1: Timing Analysis
# ----------------------------
print("Pre Synthesis:")

timing = pyrtl.TimingAnalysis()
timing.print_max_length()

# Critical path info can also be retrieved as a list of nets
critical_path_info = timing.critical_path()

# ----------------------------
# Part 2: Area Estimation
# ----------------------------
logic_area, mem_area = pyrtl.area_estimation(tech_in_nm=65)
est_area = logic_area + mem_area
print("Estimated Area of block:", est_area, "sq mm")

# ----------------------------
# Part 3: Synthesis
# ----------------------------
# Synthesize the design: breaks combinatorial logic into 1-bit operations
pyrtl.synthesize()

print("\nPre Optimization:")
timing = pyrtl.TimingAnalysis()
timing.print_max_length()

print("Logic nets:")
for net in pyrtl.working_block().logic:
    print(str(net))

# ----------------------------
# Part 4: Optimization
# ----------------------------
# Remove redundant logic and simplify circuit
_ = pyrtl.optimize()

print("\nPost Optimization:")
timing = pyrtl.TimingAnalysis()
timing.print_max_length()

print("Logic nets after optimization:")
for net in pyrtl.working_block().logic:
    print(str(net))
